const LANG_FILE = path.join(process.cwd(), "data", "telegram_user_langs.json");

function getUserLang(chatId: number): string {
  try {
    if (fs.existsSync(LANG_FILE)) {
      const data = JSON.parse(fs.readFileSync(LANG_FILE, "utf-8"));
      return data[chatId] || "en";
    }
  } catch (err) {
    console.error("Error reading user lang file:", err);
  }
  return "en";
}

function setUserLang(chatId: number, lang: string) {
  try {
    let data: any = {};
    if (fs.existsSync(LANG_FILE)) {
      data = JSON.parse(fs.readFileSync(LANG_FILE, "utf-8"));
    }
    data[chatId] = lang;
    fs.writeFileSync(LANG_FILE, JSON.stringify(data, null, 2), "utf-8");
  } catch (err) {
    console.error("Error writing user lang file:", err);
  }
}

const USAGE_FILE = path.join(process.cwd(), "data", "telegram_daily_usage.json");


const SUPPORT_MAP_FILE = path.join(process.cwd(), "data", "telegram_support_map.json");

function mapAdminMessageToUser(adminMessageId: number, userChatId: number) {
  try {
    let data: any = {};
    if (fs.existsSync(SUPPORT_MAP_FILE)) {
      data = JSON.parse(fs.readFileSync(SUPPORT_MAP_FILE, "utf-8"));
    }
    data[adminMessageId] = userChatId;
    fs.writeFileSync(SUPPORT_MAP_FILE, JSON.stringify(data, null, 2), "utf-8");
  } catch (err) {}
}

function getUserChatIdFromAdminMessage(adminMessageId: number): number | null {
  try {
    if (fs.existsSync(SUPPORT_MAP_FILE)) {
      const data = JSON.parse(fs.readFileSync(SUPPORT_MAP_FILE, "utf-8"));
      return data[adminMessageId] || null;
    }
  } catch (err) {}
  return null;
}

const SUPPORT_STATE_FILE = path.join(process.cwd(), "data", "telegram_support.json");

function getSupportState(chatId: number): { active: boolean, messages: number[] } {
  try {
    if (fs.existsSync(SUPPORT_STATE_FILE)) {
      const data = JSON.parse(fs.readFileSync(SUPPORT_STATE_FILE, "utf-8"));
      return data[chatId] || { active: false, messages: [] };
    }
  } catch (err) {}
  return { active: false, messages: [] };
}

function setSupportState(chatId: number, active: boolean, newMessages: number[] = []) {
  try {
    let data: any = {};
    if (fs.existsSync(SUPPORT_STATE_FILE)) {
      data = JSON.parse(fs.readFileSync(SUPPORT_STATE_FILE, "utf-8"));
    }
    const currentState = data[chatId] || { active: false, messages: [] };
    
    if (!active) {
      delete data[chatId];
    } else {
      data[chatId] = { active: true, messages: [...currentState.messages, ...newMessages] };
    }
    fs.writeFileSync(SUPPORT_STATE_FILE, JSON.stringify(data, null, 2), "utf-8");
  } catch (err) {}
}

async function setMessageReaction(baseUrl: string, chatId: number, messageId: number, emoji: string) {
  try {
    await fetch(`${baseUrl}/setMessageReaction`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        chat_id: chatId,
        message_id: messageId,
        reaction: [{ type: "emoji", emoji: emoji }]
      })
    });
  } catch (err) {}
}


function checkDownloadLimit(chatId: number): boolean {
  if (getUserSub(chatId)) return true; // Premium users have no limit

  try {
    if (fs.existsSync(USAGE_FILE)) {
      const data = JSON.parse(fs.readFileSync(USAGE_FILE, "utf-8"));
      const today = new Date().toISOString().split('T')[0];
      
      if (!data[today]) {
        return true;
      }
      
      const count = data[today][chatId] || 0;
      if (count >= 5) {
        return false;
      }
    }
  } catch (err) {
    console.error("Error reading usage file:", err);
  }
  return true;
}

function incrementDownloadCount(chatId: number) {
  if (getUserSub(chatId)) return; // Don't track premium users

  try {
    let data: any = {};
    if (fs.existsSync(USAGE_FILE)) {
      data = JSON.parse(fs.readFileSync(USAGE_FILE, "utf-8"));
    }
    
    const today = new Date().toISOString().split('T')[0];
    
    // Clean up old days to prevent file growing indefinitely
    const newData: any = {};
    if (data[today]) newData[today] = data[today];
    else newData[today] = {};
    
    newData[today][chatId] = (newData[today][chatId] || 0) + 1;
    
    fs.writeFileSync(USAGE_FILE, JSON.stringify(newData, null, 2), "utf-8");
  } catch (err) {
    console.error("Error writing usage file:", err);
  }
}

function getDownloadStats(chatId: number): number {
  if (getUserSub(chatId)) return 0;

  try {
    if (fs.existsSync(USAGE_FILE)) {
      const data = JSON.parse(fs.readFileSync(USAGE_FILE, "utf-8"));
      const today = new Date().toISOString().split('T')[0];
      if (data[today] && data[today][chatId]) {
        return data[today][chatId];
      }
    }
  } catch (err) {
    console.error("Error reading usage file:", err);
  }
  return 0;
}

function getReferralData(chatId: number): { count: number; days: number } {
  try {
    if (fs.existsSync(REFERRAL_FILE)) {
      const data = JSON.parse(fs.readFileSync(REFERRAL_FILE, "utf-8"));
      if (data[chatId]) {
        return data[chatId];
      }
    }
  } catch (err) {
    console.error("Error reading referral file:", err);
  }
  return { count: 0, days: 0 };
}

function addReferral(referrerId: number, inviteeId: number) {
  try {
    let data: any = {};
    if (fs.existsSync(REFERRAL_FILE)) {
      data = JSON.parse(fs.readFileSync(REFERRAL_FILE, "utf-8"));
    }
    
    if (!data.invited) {
      data.invited = {};
    }
    
    if (!data.invited[inviteeId]) {
      data.invited[inviteeId] = referrerId;
      
      if (!data[referrerId]) {
        data[referrerId] = { count: 0, days: 0 };
      }
      
      data[referrerId].count += 1;
      data[referrerId].days += 3;
      
      setUserSub(referrerId, 3);

      fs.writeFileSync(REFERRAL_FILE, JSON.stringify(data, null, 2), "utf-8");
    }
  } catch (err) {
    console.error("Error writing referral file:", err);
  }
}

const SUB_FILE = path.join(process.cwd(), "data", "telegram_user_subs.json");

function getUserSub(chatId: number): boolean {
  try {
    if (fs.existsSync(SUB_FILE)) {
      const data = JSON.parse(fs.readFileSync(SUB_FILE, "utf-8"));
      if (data[chatId] === true) {
        return true;
      }
      if (data[chatId] && data[chatId] > Date.now()) {
        return true;
      }
    }
  } catch (err) {
    console.error("Error reading user sub file:", err);
  }
  return false;
}

function setUserSub(chatId: number, durationDays: number | "lifetime") {
  try {
    let data: any = {};
    if (fs.existsSync(SUB_FILE)) {
      data = JSON.parse(fs.readFileSync(SUB_FILE, "utf-8"));
    }
    if (durationDays === "lifetime") {
      data[chatId] = true;
    } else {
      const currentExpiry = (data[chatId] && data[chatId] > Date.now()) ? data[chatId] : Date.now();
      data[chatId] = currentExpiry + (durationDays * 24 * 60 * 60 * 1000);
    }
    fs.writeFileSync(SUB_FILE, JSON.stringify(data, null, 2), "utf-8");
  } catch (err) {
    console.error("Error writing user sub file:", err);
  }
}

const HISTORY_FILE = path.join(process.cwd(), "data", "telegram_history.json");

interface HistoryEntry {
  title: string;
  url: string;
  date: string;
}

function getUserHistory(chatId: number): HistoryEntry[] {
  try {
    if (fs.existsSync(HISTORY_FILE)) {
      const data = JSON.parse(fs.readFileSync(HISTORY_FILE, "utf-8"));
      return data[chatId] || [];
    }
  } catch (err) {
    console.error("Error reading history file:", err);
  }
  return [];
}

function addUserHistory(chatId: number, title: string, url: string) {
  try {
    let data: Record<number, HistoryEntry[]> = {};
    if (fs.existsSync(HISTORY_FILE)) {
      data = JSON.parse(fs.readFileSync(HISTORY_FILE, "utf-8"));
    }
    if (!data[chatId]) {
      data[chatId] = [];
    }
    const entry: HistoryEntry = {
      title,
      url,
      date: new Date().toISOString()
    };
    data[chatId].unshift(entry);
    // Keep max 10 entries
    if (data[chatId].length > 10) {
      data[chatId].pop();
    }
    fs.writeFileSync(HISTORY_FILE, JSON.stringify(data, null, 2), "utf-8");
  } catch (err) {
    console.error("Error writing history file:", err);
  }
}

function clearUserHistory(chatId: number) {
  try {
    let data: any = {};
    if (fs.existsSync(HISTORY_FILE)) {
      data = JSON.parse(fs.readFileSync(HISTORY_FILE, "utf-8"));
    }
    if (data[chatId]) {
      delete data[chatId];
      fs.writeFileSync(HISTORY_FILE, JSON.stringify(data, null, 2), "utf-8");
    }
  } catch (err) {
    console.error("Error clearing history file:", err);
  }
}
